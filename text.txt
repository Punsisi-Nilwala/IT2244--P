01.print multiplication table using for loop


multiplier=2


for ((i=1; i<=10; i++))
do
    echo "$i * $multiplier = $((i * multiplier))"
done

03. sum of fibonnaci numbers


echo "Enter the number of Fibonacci terms:"
read n


a=0
b=1
sum=0


for (( i=0; i<n; i++ ))
do
    echo -n "$a "  
    sum=$((sum + a))  
    temp=$((a + b))   
    a=$b
    b=$temp
done


echo -e "\nSum of Fibonacci series up to $n terms: $sum"

        fi
