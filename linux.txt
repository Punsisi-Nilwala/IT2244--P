Q1)Print multiplication table using for loop.
1 X 2 = 2
2 X 2 = 4

//Code
{ ~ }  » vi QQ8.sh 
echo 'Multiplication Table'
echo 'Enter number: '
read num

for ((i=1; i<=10; i++))
do
a=$(($i*$num))
echo -n  $i ' X ' $num ': ' $a
echo ''
done

Q3)sum of the first 10 fibbonacci series

//Code
{ ~ }  » vi QQ10.sh  
a=0
b=1

for (( i=0; i<10; i++))
do echo $a
echo ""
c=$(($a+$b))
sum=$(($sum+$a))

b=$c
done

Q4)sum of prime numbers from 1 to 100

//code
{ ~ }  » vi QQ11.sh 
sum=0

for (( num=2; num<=100; num++ ))
do
    is_prime=1
    for (( i=2; i*i<=num; i++ ))
    do
        if (( num % i == 0 )); then
            is_prime=0
            break
        fi
    done

    if (( is_prime == 1 )); then
        sum=$((sum + num))
    fi
done

echo "Sum of prime numbers from 1 to 100 is: $sum"


//output

{ ~ }  » ./QQ11.sh                                                            ~
